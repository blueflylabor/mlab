DEF sub_o(SCALE:IN, APPHEIGHT:IN)
DECL REAL SCALE, APPHEIGHT
DECL POS P, PZ

sub_move_z(-APPHEIGHT)
P.X = 19.5*SCALE
P.Y = 0*SCALE
PTP_REL P

sub_move_z(APPHEIGHT)
P.X = 19.5*SCALE
P.Y = 19.5*SCALE
PZ.X = 0*SCALE
PZ.Y = 39*SCALE
CIRC_REL P, PZ

P.X = -19.5*SCALE
P.Y = -19.5*SCALE
PZ.X = 0*SCALE
PZ.Y = -39*SCALE
CIRC_REL P, PZ


sub_move_z(-APPHEIGHT)
P.X = 0*SCALE
P.Y = 8*SCALE
PTP_REL P

sub_move_z(APPHEIGHT)
P.X = 11.5*SCALE
P.Y = 11.5*SCALE
PZ.X = 0*SCALE
PZ.Y = 23*SCALE
CIRC_REL P, PZ

P.X = -11.5*SCALE
P.Y = -11.5*SCALE
PZ.X = 0*SCALE
PZ.Y = -23*SCALE
CIRC_REL P, PZ

sub_move_z(-APPHEIGHT)

END
